# Runtime dependencies for the sandbox SEC graph engine and its query server.
#
# Pinned to the versions this was built and verified against. Regenerate with:
#   .venv/bin/python -m pip freeze > requirements.txt
#
# Python 3.13 (the LadybugDB and pandas wheels below are 3.13-built).

# Graph store. LadybugDB is the embedded property-graph engine behind
# sandbox_engine/ddl.py, loader.py and benchmarks.py.
ladybug==0.20.4

# Parse stage: pandas for table extraction, lxml for read_html flavor="lxml",
# beautifulsoup4 for strip_markup()'s visibility-aware text extraction.
pandas==3.0.6
lxml==6.1.3
beautifulsoup4==4.15.0

# Staging: Arrow/Parquet spill between the parse and load stages.
pyarrow==25.0.1

# PDF text extraction. graphrag/document.py imports PdfReader at module scope,
# so a missing pypdf breaks `import graphrag` entirely -- which takes the
# query UI and the whole legacy path down with it, not just PDF handling.
# document_loader.py also prefers pdfplumber for table recovery, but degrades
# to pypdf-only without it, so it is not pinned.
pypdf==6.19.0

# Query server: the OpenAI-compatible client used for GraphRAG answers, and
# typer for the CLI entry point in sandbox_engine/cli.py.
openai==3.19.2
typer==0.27.2

# Cold-start JIT Graph RAG: Pydantic schema validation and in-memory overlay graph.
pydantic>=2.0.0
networkx>=3.0
